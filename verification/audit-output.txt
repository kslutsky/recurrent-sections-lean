def RecurrentSections.PolynomialGrowth : (ℕ → ℕ) → Prop :=
fun v => ∃ C d, ∀ (n : ℕ), v n ≤ C * (n + 1) ^ d
def RecurrentSections.TwoSidedPolynomialGrowth : (ℕ → ℕ) → Prop :=
fun v => ∃ C d, 0 < C ∧ ∀ (n : ℕ), (n + 1) ^ d ≤ C * v n ∧ v n ≤ C * (n + 1) ^ d

def RecurrentSections.UniversalMaximalRecurrence : {G : Type} →
  [inst : Group G] → [inst_1 : DecidableEq G] → RecurrentSections.WordGeometry G → Prop :=
fun {G} [Group G] [DecidableEq G] W =>
  ∀ (X : Type) [inst_2 : MeasurableSpace X] [StandardBorelSpace X] [inst_4 : MulAction G X] [MeasurableConstSMul G X],
    RecurrentSections.HasRecurrentMaximalSections W
def RecurrentSections.HasRecurrentMaximalSections : {G X : Type} →
  [inst : Group G] →
    [inst_1 : DecidableEq G] → [MulAction G X] → [MeasurableSpace X] → RecurrentSections.WordGeometry G → Prop :=
fun {G X} [Group G] [DecidableEq G] [MulAction G X] [MeasurableSpace X] W =>
  ∀ (r : ℕ → ℕ),
    StrictMono r →
      (∀ (n : ℕ), 0 < r n) →
        ∃ C,
          (∀ (n : ℕ), MeasurableSet (C n)) ∧
            (∀ (n : ℕ), RecurrentSections.MaximalSeparated W (r n) (C n)) ∧
              (∀ (n : ℕ), RecurrentSections.CompleteSection (C n)) ∧ RecurrentSections.Recurrent W r C
